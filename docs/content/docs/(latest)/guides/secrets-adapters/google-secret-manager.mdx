## Add the adapter

Hot Updater ships no Secret Manager adapter: paste this file into your
project. It implements [`SecretsAdapter`](/docs/concepts/secrets#write-a-secrets-adapter)
with Google's official client, keeping each name as the secret
`<prefix><name>` and reading its `latest` version.

```package-install
npm install @google-cloud/secret-manager --save-dev
```

```ts title="secrets/google-secret-manager.ts"
import { SecretManagerServiceClient } from "@google-cloud/secret-manager";
import type { SecretsAdapter } from "hot-updater";

/** gRPC's NOT_FOUND. */
const NOT_FOUND = 5;

const isNotFound = (error: unknown) =>
  (error as { code?: unknown } | null)?.code === NOT_FOUND;

export const googleSecretManager = ({
  project,
  prefix = "hot-updater-",
}: {
  project: string;
  prefix?: string;
}): SecretsAdapter => {
  // Credentials come from Application Default Credentials: gcloud on your
  // machine, Workload Identity Federation in CI.
  const client = new SecretManagerServiceClient();
  // Secret IDs take letters, digits, "-", and "_".
  const secret = (name: string) =>
    `projects/${project}/secrets/${prefix}${name.replace(/[^\w-]/g, "_")}`;

  const read = async (name: string) => {
    try {
      const [version] = await client.accessSecretVersion({
        name: `${secret(name)}/versions/latest`,
      });
      const data = version.payload?.data;
      return data ? Buffer.from(data).toString("utf-8") : undefined;
    } catch (error) {
      if (isNotFound(error)) return undefined;
      throw error;
    }
  };

  return {
    name: "googleSecretManager",
    get: async (names) =>
      Object.fromEntries(
        await Promise.all(
          names.map(async (name) => [name, await read(name)] as const),
        ),
      ),
    set: async (name, value) => {
      const payload = { data: Buffer.from(value, "utf-8") };
      try {
        await client.addSecretVersion({ parent: secret(name), payload });
      } catch (error) {
        if (!isNotFound(error)) throw error;
        await client.createSecret({
          parent: `projects/${project}`,
          secretId: secret(name).split("/").at(-1),
          secret: { replication: { automatic: {} } },
        });
        await client.addSecretVersion({ parent: secret(name), payload });
      }
    },
    describe: (name) => `Secret Manager secret ${secret(name)}`,
  };
};
```

```ts title="hot-updater.config.ts (partial)"
import { defineConfig } from "hot-updater";

import { googleSecretManager } from "./secrets/google-secret-manager";

export default defineConfig({
  secrets: googleSecretManager({ project: "my-project" }),
  // build, storage, database and plugins stay as they are
});
```

With the default prefix, `secret:asc-api-key` reads the secret
`hot-updater-asc-api-key`. A credential is one
[JSON value](/docs/concepts/secrets#sign-native-builds-with-secrets) per
secret.

The client signs in with Application Default Credentials:
`gcloud auth application-default login` on your machine, and the Workload
Identity Federation file `google-github-actions/auth` writes in CI.

## Grant least privilege

| Who                                                    | Role                                                                                  |
| ------------------------------------------------------ | ------------------------------------------------------------------------------------- |
| A CI job that builds or deploys                        | `roles/secretmanager.secretAccessor` on each secret it reads                          |
| The developer who runs `secrets put` or signs in       | `roles/secretmanager.secretVersionAdder` and `secretAccessor`, and `secrets.create` on the project for a secret that does not exist yet |

Each `set` adds a version. Secret Manager's
[free tier](https://cloud.google.com/secret-manager/pricing) covers six
active versions: destroy the old ones once nothing reads them, such as with
`gcloud secrets versions destroy`.

## Read secrets in GitHub Actions

Create a Workload Identity pool and a provider for your repository, and let
the CI service account read the build's secrets:

```bash
gcloud iam workload-identity-pools create github --project=my-project \
  --location=global
gcloud iam workload-identity-pools providers create-oidc my-app \
  --project=my-project --location=global --workload-identity-pool=github \
  --issuer-uri=https://token.actions.githubusercontent.com \
  --attribute-mapping=google.subject=assertion.sub,attribute.repository=assertion.repository \
  --attribute-condition="assertion.repository == 'my-org/my-app'"
gcloud iam service-accounts create hot-updater-ci --project=my-project
gcloud iam service-accounts add-iam-policy-binding \
  hot-updater-ci@my-project.iam.gserviceaccount.com \
  --role=roles/iam.workloadIdentityUser \
  --member="principalSet://iam.googleapis.com/projects/PROJECT_NUMBER/locations/global/workloadIdentityPools/github/attribute.repository/my-org/my-app"
for secret in hot-updater-release-keystore hot-updater-play-service-account; do
  gcloud secrets add-iam-policy-binding "$secret" --project=my-project \
    --role=roles/secretmanager.secretAccessor \
    --member=serviceAccount:hot-updater-ci@my-project.iam.gserviceaccount.com
done
```

The job asks GitHub for an OIDC token (`id-token: write`), and
`google-github-actions/auth` writes the federation file the client reads:

```yaml title=".github/workflows/native-release.yml (partial)"
permissions:
  contents: read
  id-token: write

# ...in the job, after npm ci:
- uses: google-github-actions/auth@7c6bc770dae815cd3e89ee6cdf493a5fab2cc093 # v3
  with:
    workload_identity_provider: projects/PROJECT_NUMBER/locations/global/workloadIdentityPools/github/providers/my-app
    service_account: hot-updater-ci@my-project.iam.gserviceaccount.com
- uses: gronxb/hot-updater/actions/native-build@v1.2.0
  with:
    platform: android
    scheme: release
    upload: play-internal
```

has the complete workflow.
